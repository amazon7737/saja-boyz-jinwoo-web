async function until(check, milliseconds, message) {
  const deadline = performance.now() + milliseconds;
  while (!check()) {
    assert(performance.now() < deadline, message);
    await delay(50);
  }
}
async function freePort() {
  const reserve = net.createServer();
  await new Promise((resolve, reject) => { reserve.once('error', reject); reserve.listen(0, '127.0.0.1', resolve); });
  const port = reserve.address().port;
  await new Promise((resolve) => reserve.close(resolve));
  return port;
}
function startChild(args, cwd, env, label) {
  checkActive();
  const output = fs.openSync(path.join(runDir, `${label}.log`), 'w', 0o600);
  const child = spawn(process.execPath, args, { cwd, env, detached: true, stdio: ['ignore', output, output] });
  fs.closeSync(output);
  children.add(child);
  // Retain the process group until final cleanup, even if its leader exits.
  return child;
}
async function stopChild(child) {
  if (!child.pid) return;
  const exists = () => {
    try { process.kill(-child.pid, 0); return true; }
    catch (error) { if (error.code === 'ESRCH') return false; throw error; }
  };
  if (!exists()) return;
  try { process.kill(-child.pid, 'SIGTERM'); } catch (error) { if (error.code !== 'ESRCH') throw error; }
  const deadline = performance.now() + 1500;
  while (exists() && performance.now() < deadline) await delay(50, false);
  if (exists()) {
    try { process.kill(-child.pid, 'SIGKILL'); } catch (error) { if (error.code !== 'ESRCH') throw error; }
  }
}
async function finishChild(child, timeout) {
  let timer;
  let onAbort;
  try {
    return await Promise.race([
      new Promise((resolve, reject) => {
        child.once('error', reject);
        child.once('exit', (code, signal) => code === 0 ? resolve() : reject(new Error(`Child failed: ${code ?? signal}`)));
      }),
      new Promise((_, reject) => { timer = setTimeout(() => reject(new Error('Isolated build timed out')), timeout); }),
      new Promise((_, reject) => {
        onAbort = () => reject(operationAbort.signal.reason);
        if (operationAbort.signal.aborted) onAbort();
        else operationAbort.signal.addEventListener('abort', onAbort, { once: true });
      }),
    ]);
  } finally {
    clearTimeout(timer);
    if (onAbort) operationAbort.signal.removeEventListener('abort', onAbort);
  }
}
async function serveRequest(request, response) {
  response.on('error', (error) => { evidence.mock_write_error = error.code || error.name; });
  if (request.url !== '/reports') { response.writeHead(404); response.end(); return; }
  if (request.headers.origin !== frontendOrigin) { response.writeHead(403); response.end(); return; }
  const cors = { 'Access-Control-Allow-Origin': frontendOrigin, Vary: 'Origin' };
  if (request.method === 'OPTIONS') {
    evidence.options.push({ method: request.headers['access-control-request-method'], headers: request.headers['access-control-request-headers'], origin: request.headers.origin });
    response.writeHead(204, { ...cors, 'Access-Control-Allow-Methods': 'POST', 'Access-Control-Allow-Headers': 'Content-Type, Idempotency-Key' });
    response.end();
    return;
  }
  if (request.method !== 'POST') { response.writeHead(405); response.end(); return; }
  const record = {
    number: evidence.posts.length + 1,
    received_at: new Date().toISOString(),
    arrived_monotonic_ms: performance.now(),
    idempotency_key: request.headers['idempotency-key'],
    origin: request.headers.origin,
    content_type: request.headers['content-type'],
    has_cookie: Boolean(request.headers.cookie),
  };
  evidence.posts.push(record);
  response.on('close', () => {
    record.connection_closed_after_ms = performance.now() - record.arrived_monotonic_ms;
    record.response_finished = response.writableFinished;
  });
  const chunks = [];
  let byteCount = 0;
  for await (const chunk of request) {
    byteCount += chunk.length;
    assert(byteCount < 1024 * 1024, 'Diagnostic uploads must stay below 1 MiB');
    chunks.push(chunk);
  }
  const form = await new Request('http://127.0.0.1/reports', { method: 'POST', headers: { 'Content-Type': record.content_type }, body: Buffer.concat(chunks) }).formData();
  record.fields = {};
  record.files = [];
  for (const [name, value] of form) {
    if (typeof value === 'string') record.fields[name] = value;
    else record.files.push({ field: name, name: value.name, type: value.type, bytes: value.size, sha256: hash(Buffer.from(await value.arrayBuffer())) });
  }
  record.parsed = true;
  if (record.number === 1) {
    firstResponse = response;
    const timer = setTimeout(() => {
      ownedTimers.delete(timer);
      if (response.destroyed) return;
      response.writeHead(200, { ...cors, 'Content-Type': 'application/json', 'Cache-Control': 'no-store' });
      response.flushHeaders();
      response.write('{"report_id":');
      record.headers_sent_after_ms = performance.now() - record.arrived_monotonic_ms;
      // Deliberately do not end the body. Native browser Fetch must abort it.
    }, 45000);
    ownedTimers.add(timer);
    response.once('close', () => { clearTimeout(timer); ownedTimers.delete(timer); });
    return;
  }
  assert.equal(record.number, 2, 'There must be one explicit retry, never an automatic resend');
  response.writeHead(200, { ...cors, 'Content-Type': 'application/json', 'Cache-Control': 'no-store' });
  response.end(JSON.stringify(receipt));
}
async function ready(url, child) {
  const deadline = performance.now() + 30000;
  while (performance.now() < deadline) {
    assert(child.exitCode === null && child.signalCode === null, 'Next exited before readiness');
    try {
      const response = await fetch(url, { signal: AbortSignal.timeout(1000) });
      if (response.status === 200) { await response.arrayBuffer(); return; }
    } catch { /* Local listener may not yet be ready. */ }
    await delay(100);
  }
  throw new Error('Next did not become ready');
}
